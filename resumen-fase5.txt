import type { z } from 'zod';

import type { Prisma } from '@/generated/prisma/client';
import { registrarAuditoria, registrarAuditoriaTx } from '@/shared/audit';
import { getSession } from '@/shared/auth/session';
import type { SessionUser } from '@/shared/auth/types';
import { withTransaction } from '@/shared/db/transaction';
import { ERROR_CODES } from '@/shared/errors/codes';
import { type ErrorContext, handleError } from '@/shared/errors/handler';
import { fail, ok, type Result } from '@/shared/errors/result';
import { runWithTenant } from '@/shared/tenant/context';

import type { ActionContext, ValidatedActionConfig } from './types';

type ActionHandler<TSchema extends z.ZodTypeAny, TOut> = (
  input: z.infer<TSchema>,
  ctx: ActionContext,
) => Promise<TOut>;

/**
 * Envuelve una Server Action aplicando, en orden:
 *   1. Validación Zod del input
 *   2. Autenticación (getSession)
 *   3. Autorización (permission y/o roles)
 *   4. Aislamiento multi-tenant (runWithTenant)
 *   5. Transacción ACID (si transactional: true)
 *   6. Auditoría (si config.audit)
 *   7. Manejo centralizado de errores → Result<T>
 *
 * NUNCA lanza. Siempre devuelve Result<T>.
 */
export function validatedAction<TSchema extends z.ZodTypeAny, TOut>(
  config: ValidatedActionConfig<TSchema, TOut>,
  handler: ActionHandler<TSchema, TOut>,
): (rawInput: unknown) => Promise<Result<TOut>> {
  const errorCtx: ErrorContext = { modulo: config.modulo, accion: config.accion };

  return async (rawInput: unknown): Promise<Result<TOut>> => {
    // ─── 1. Validación Zod ───
    const parsed = config.schema.safeParse(rawInput);
    if (!parsed.success) {
      return handleError(parsed.error, errorCtx);
    }
    const input = parsed.data;

    // ─── 2. Autenticación ───
    let user: SessionUser | null = null;
    try {
      user = await getSession();
    } catch (err) {
      return handleError(err, errorCtx);
    }
    if (!user) {
      return fail(
        ERROR_CODES.UNAUTHENTICATED,
        `[${config.modulo}:${config.accion}] Debes iniciar sesión para continuar`,
      );
    }

    // ─── 3. Autorización ───
    if (config.permission && !user.permisos.includes(config.permission)) {
      return fail(
        ERROR_CODES.FORBIDDEN,
        `[${config.modulo}:${config.accion}] Falta el permiso requerido: ${config.permission}`,
      );
    }
    if (config.roles && !config.roles.some((r) => user.roles.includes(r))) {
      return fail(
        ERROR_CODES.FORBIDDEN,
        `[${config.modulo}:${config.accion}] Se requiere uno de los roles: ${config.roles.join(', ')}`,
      );
    }

    // ─── 4. Ejecución dentro del contexto del tenant ───
    const effectiveUser = user;
    return runWithTenant(
      { tenantId: effectiveUser.tenantId, usuarioId: effectiveUser.id },
      async (): Promise<Result<TOut>> => {
        try {
          if (config.transactional) {
            return await runTransactional(config, handler, input, effectiveUser);
          }
          return await runStandalone(config, handler, input, effectiveUser);
        } catch (err) {
          return handleError(err, {
            ...errorCtx,
            usuarioId: effectiveUser.id,
            tenantId: effectiveUser.tenantId,
          });
        }
      },
    );
  };
}

/**
 * Ejecuta handler + auditoría dentro de una única transacción ACID.
 */
async function runTransactional<TSchema extends z.ZodTypeAny, TOut>(
  config: ValidatedActionConfig<TSchema, TOut>,
  handler: ActionHandler<TSchema, TOut>,
  input: z.infer<TSchema>,
  user: SessionUser,
): Promise<Result<TOut>> {
  return withTransaction(async (tx) => {
    const ctx: ActionContext = {
      user,
      tenantId: user.tenantId,
      tx,
    };

    const out = await handler(input, ctx);

    // Auditoría DENTRO de la transacción → si falla, rollback completo.
    if (config.audit) {
      const datosAnteriores = config.audit.datosAnteriores
        ? await config.audit.datosAnteriores(input, { tx })
        : null;
      const datosNuevos = config.audit.datosNuevos ? config.audit.datosNuevos(out) : null;

      await registrarAuditoriaTx(tx, {
        modulo: config.modulo,
        accion: config.audit.accionAuditoria,
        tablaAfectada: config.audit.tablaAfectada,
        registroId: config.audit.registroId(out),
        datosAnteriores,
        datosNuevos,
      });
    }

    return out;
  }).then(
    (data) => ok(data),
    (err) =>
      handleError(err, {
        modulo: config.modulo,
        accion: config.accion,
        usuarioId: user.id,
        tenantId: user.tenantId,
      }),
  );
}

/**
 * Ejecuta handler + auditoría sin transacción.
 */
async function runStandalone<TSchema extends z.ZodTypeAny, TOut>(
  config: ValidatedActionConfig<TSchema, TOut>,
  handler: ActionHandler<TSchema, TOut>,
  input: z.infer<TSchema>,
  user: SessionUser,
): Promise<Result<TOut>> {
  const ctx: ActionContext = {
    user,
    tenantId: user.tenantId,
    tx: null,
  };

  const out = await handler(input, ctx);

  if (config.audit) {
    const datosAnteriores = config.audit.datosAnteriores
      ? await config.audit.datosAnteriores(input, undefined)
      : null;
    const datosNuevos = config.audit.datosNuevos ? config.audit.datosNuevos(out) : null;

    await registrarAuditoria({
      modulo: config.modulo,
      accion: config.audit.accionAuditoria,
      tablaAfectada: config.audit.tablaAfectada,
      registroId: config.audit.registroId(out),
      datosAnteriores,
      datosNuevos,
    });
  }

  return ok(out);
}

import type { ErrorCode } from './codes';

/**
 * Tipo Result<T>: representa el resultado de una operación que puede fallar.
 * Reemplaza el uso de excepciones en Server Actions.
 *
 * Uso:
 *   const result = await algunaAction(input);
 *   if (!result.success) {
 *     console.error(result.code, result.error);
 *     return;
 *   }
 *   console.log(result.data); // tipado como T
 */
export type Result<T> =
  | { success: true; data: T }
  | { success: false; error: string; code: ErrorCode; details?: unknown };

/** Crea un resultado exitoso. */
export function ok<T>(data: T): Result<T> {
  return { success: true, data };
}

/** Crea un resultado fallido. */
export function fail(code: ErrorCode, error: string, details?: unknown): Result<never> {
  return { success: false, code, error, details };
}

/** Type guard: comprueba si un Result es exitoso. */
export function isOk<T>(result: Result<T>): result is { success: true; data: T } {
  return result.success;
}

/** Type guard: comprueba si un Result es fallido. */
export function isFail<T>(
  result: Result<T>,
): result is { success: false; error: string; code: ErrorCode; details?: unknown } {
  return !result.success;
}

import type { Prisma } from '@/generated/prisma/client';
import { prisma } from '@/shared/db/client';
import { BusinessRuleError, ConflictError, NotFoundError } from '@/shared/errors';
import { requireCurrentTenantId } from '@/shared/tenant/context';

import type {

// ═══════════════════════════════════════════════════════════════
// CREAR
// ═══════════════════════════════════════════════════════════════

export async function crearDiagnostico(
  input: CrearDiagnosticoInput,
  usuarioId: string,
  ctx: Ctx = {},
) {
  const client = ctx.tx ?? prisma;
  const tenantId = requireCurrentTenantId();

  // Verificar que la orden existe y está en estado válido
  const orden = await client.ordenServicio.findFirst({
    where: { id: input.ordenId },
    select: { id: true, estado: true },
  });
  if (!orden) {
    throw new NotFoundError('Orden no encontrada');
  }

  const estadosPermitidos = ['RECIBIDO', 'EN_DIAGNOSTICO'];
  if (!estadosPermitidos.includes(orden.estado)) {
    throw new BusinessRuleError(
      `No se puede crear un diagnóstico en una orden en estado "${orden.estado}"`,
    );
  }

  // Verificar que no exista un diagnóstico activo (BORRADOR o EN_PROCESO)
  const activo = await client.diagnostico.findFirst({
    where: {
      ordenId: input.ordenId,
      estado: { in: ['BORRADOR', 'EN_PROCESO'] },
    },
    select: { id: true },
  });
  if (activo) {
    throw new ConflictError('Ya existe un diagnóstico activo para esta orden');
  }

  const data: Prisma.DiagnosticoUncheckedCreateInput = {
    tenantId,
    ordenId: input.ordenId,
    tecnicoId: usuarioId,
    estado: 'EN_PROCESO',
    pruebasRealizadas: input.pruebasRealizadas ?? null,
    voltajesMedidos: input.voltajesMedidos ?? null,
    componentesDanados: input.componentesDanados ?? null,
    observaciones: input.observaciones ?? null,
    fechaInicio: new Date(),
  };

  return client.diagnostico.create({ data });
}

// ═══════════════════════════════════════════════════════════════
// EDITAR
// ═══════════════════════════════════════════════════════════════

export async function editarDiagnostico(input: EditarDiagnosticoInput, ctx: Ctx = {}) {
  const client = ctx.tx ?? prisma;

  const actual = await client.diagnostico.findFirst({
    where: { id: input.id },
    select: { id: true, estado: true },
  });
  if (!actual) {
    throw new NotFoundError('Diagnóstico no encontrado');
  }

  if (actual.estado === 'COMPLETADO') {
    throw new BusinessRuleError('No se puede editar un diagnóstico completado');
  }

  const data: Prisma.DiagnosticoUncheckedUpdateInput = {};
  if (input.pruebasRealizadas !== undefined) {
    data.pruebasRealizadas = input.pruebasRealizadas ?? null;
  }
  if (input.voltajesMedidos !== undefined) {
    data.voltajesMedidos = input.voltajesMedidos ?? null;
  }
  if (input.componentesDanados !== undefined) {
    data.componentesDanados = input.componentesDanados ?? null;
  }
  if (input.observaciones !== undefined) {
    data.observaciones = input.observaciones ?? null;
  }
  if (input.conclusion !== undefined) {
    data.conclusion = input.conclusion ?? null;
  }

  return client.diagnostico.update({
    where: { id: input.id },
    data,
  });
}

// COMPLETAR (bloquea edición futura)

export async function completarDiagnostico(input: CompletarDiagnosticoInput, ctx: Ctx = {}) {
  const client = ctx.tx ?? prisma;

  const actual = await client.diagnostico.findFirst({
    where: { id: input.id },
    select: { id: true, estado: true },
  });
  if (!actual) {
    throw new NotFoundError('Diagnóstico no encontrado');
  }

  if (actual.estado === 'COMPLETADO') {
    throw new BusinessRuleError('El diagnóstico ya está completado');
  }

  return client.diagnostico.update({
    where: { id: input.id },
    data: {
      estado: 'COMPLETADO',
      conclusion: input.conclusion,
      fechaFin: new Date(),
    },
  });
}

export async function obtenerDiagnostico(id: string): Promise<DiagnosticoConRelaciones | null> {

    where: { id },
    include: {
      tecnico: { select: { id: true, nombre: true, apellidos: true } },
      evidencias: { orderBy: { createdAt: 'desc' } },
    },
  });

  return diag as DiagnosticoConRelaciones | null;
}
